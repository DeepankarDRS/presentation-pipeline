// tests/compile/run-all.js
// Compiles every test-*.xml in this directory and reports pass/fail.
// Usage: node tests/compile/run-all.js

import { execSync } from "child_process";
import { readdirSync } from "fs";
import { resolve, dirname, basename } from "path";
import { fileURLToPath } from "url";

const __dirname = dirname(fileURLToPath(import.meta.url));
const projectRoot = resolve(__dirname, "../..");
const compileScript = resolve(projectRoot, "node/compile-pom.js");
const outputBase = resolve(projectRoot, "output/test");

const files = readdirSync(__dirname)
  .filter((f) => f.startsWith("test-") && f.endsWith(".xml"))
  .sort();

let passed = 0;
let failed = 0;
const results = [];

for (const file of files) {
  const inputPath = resolve(__dirname, file);
  const testName = basename(file, ".xml");
  const outDir = resolve(outputBase, testName);

  try {
    const out = execSync(
      `node "${compileScript}" "${inputPath}" "${outDir}"`,
      { cwd: projectRoot, timeout: 30000, encoding: "utf-8" }
    );
    passed++;
    results.push({ test: testName, status: "PASS", output: out.trim() });
    console.log(`  PASS  ${testName}`);
  } catch (err) {
    failed++;
    const stderr = err.stderr?.toString().trim() || err.stdout?.toString().trim() || err.message;
    results.push({ test: testName, status: "FAIL", output: stderr });
    console.log(`  FAIL  ${testName}`);
    console.log(`        ${stderr.split("\n")[0]}`);
  }
}

console.log(`\n${passed} passed, ${failed} failed, ${files.length} total`);
process.exit(failed > 0 ? 1 : 0);
